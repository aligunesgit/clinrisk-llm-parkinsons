def policy(ctx, prev_r, prev_a):
    budget = ctx.get('budget', 0.0)
    sens = ctx.get('sens')
    spec = ctx.get('spec')
    trend = ctx.get('trend')

    # Thresholds (tunable, conservative on the specificity side per the spec)
    SPEC_COLLAPSE = 0.80   # below this, specificity has collapsed
    SPEC_HEADROOM = 0.90   # above this, there is room to push sensitivity harder
    SENS_POOR = 0.75       # below this, sensitivity is considered poor
    FREEZE_BUDGET = 0.85   # stop moving the risk weight past this fraction of budget

    # Cold start: too few labelled observations per class for a trustworthy
    # OOF estimate -> stay symmetric and don't bias querying either way.
    if sens is None or spec is None:
        return 1, 1

    late_stage = budget >= FREEZE_BUDGET
    r = prev_r
    a = prev_a

    if spec < SPEC_COLLAPSE:
        # Rule 3: specificity collapse overrides everything else. Back off
        # the risk weight immediately (unless frozen) and lean on targeted
        # querying of likely false negatives instead.
        if not late_stage:
            r = max(1, prev_r - 1)
        a = min(4, prev_a + 1)

        declining = (trend == 'declining')
        improving = (trend == 'improving')

        if sens_poor or declining:
            if spec >= SPEC_HEADROOM and not late_stage:
                # Rule 2: adequate specificity headroom + evidence sensitivity
                # is poor/declining -> escalate pressure gradually (one step).
                r = min(4, prev_r + 1)
                # Reweighting is doing the work now, ease query bias back
                # toward neutral so pressure isn't applied twice.
                a = max(1, prev_a - 1)
            else:
                # Sensitivity trouble but no safe headroom (or late in the
                # budget, where r is frozen): rely on targeted querying only.
                a = min(4, prev_a + 1)
        elif improving and spec >= SPEC_HEADROOM:
            # Things are healthy: gently relax query pressure, hold r steady
            # to avoid oscillation.
            a = max(1, prev_a - 1)
        # else: flat trend, acceptable spec, sens not poor -> hold (r, a)
        # unchanged to avoid oscillating actions.

    # Rule 4: freeze the risk weight late in the budget for a stable final model.
    if late_stage:
        r = prev_r

    r = int(min(4, max(1, round(r))))
    a = int(min(4, max(1, round(a))))

def policy(ctx, prev_r, prev_a):
    sens = ctx['sens']
    spec = ctx['spec']
    budget = ctx['budget']
    trend = ctx['trend']
    neg_frac = ctx['neg_frac']

    # Cold start (fewer than 3 obs/class -> sens/spec unavailable): no reliable
    # signal yet, stay fully symmetric (protect specificity by default).
    if sens is None or spec is None:
        return 1, 1

    r = prev_r
    a = prev_a

    # Hysteresis bands (separate up/down thresholds) so decisions don't flip-flop.
    SPEC_COLLAPSE = 0.85    # below this, specificity has collapsed
    SPEC_HEADROOM = 0.92    # above this, safe to push sensitivity harder
    SPEC_SAFE_DOWN = 0.90   # minimum spec required before easing risk weight back down
    SENS_POOR = 0.80        # sensitivity considered poor below this
    SENS_GOOD = 0.88        # sensitivity considered comfortably good above this
    LATE_BUDGET = 0.85      # fraction of budget past which r is frozen for stability
    MIN_NEG_POOL = 0.05     # need some negative-predicted pool for querying to help

    sens_poor = (sens < SENS_POOR) or (trend == "declining")
    sens_good = (sens >= SENS_GOOD) and (trend != "declining")

    if spec < SPEC_COLLAPSE:
        # Rule 3: specificity collapse -> back off risk weight immediately and
        # lean on targeted querying of likely false negatives instead.
        r = max(1, prev_r - 1)
        if neg_frac >= MIN_NEG_POOL:
            a = min(4, prev_a + 1)
    elif budget >= LATE_BUDGET:
        # Rule 4: stop moving risk weight late in the budget; only 'a' may adjust.
        r = prev_r
        if sens_poor and spec >= SPEC_SAFE_DOWN and neg_frac >= MIN_NEG_POOL:
            a = min(4, prev_a + 1)
        elif sens_good and prev_a > 1:
            a = max(1, prev_a - 1)
    else:
        if spec >= SPEC_HEADROOM and sens_poor:
            # Rule 2: adequate specificity headroom + poor/declining sensitivity
            # -> escalate pressure gradually (one step at a time).
            r = min(4, prev_r + 1)
        elif spec < SPEC_HEADROOM and sens_poor:
            # Sensitivity is poor but no headroom to raise risk weight safely ->
            # use targeted querying of possible false negatives instead.
            if neg_frac >= MIN_NEG_POOL:
                a = min(4, prev_a + 1)
        elif sens_good and spec >= SPEC_SAFE_DOWN and prev_r > 1:
            # Sensitivity recovered and specificity safe -> gradually ease back
            # towards symmetric training to avoid overcorrection.
            r = max(1, prev_r - 1)
            if prev_a > 1:
                a = max(1, prev_a - 1)
        # else: hold steady (inside hysteresis band, avoid oscillation)

    return int(r), int(a)
